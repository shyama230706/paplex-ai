def extract_text(uploaded_file):
    name = uploaded_file.name.lower()
    text = ""
    try:
        if name.endswith(".pdf"):
            from pypdf import PdfReader
            reader = PdfReader(BytesIO(uploaded_file.read()))
            for page in reader.pages:
                text += page.extract_text() or ""
        elif name.endswith(".txt"):
            text = uploaded_file.read().decode("utf-8", errors="ignore")
        elif name.endswith(".csv"):
            import pandas as pd
            df = pd.read_csv(uploaded_file)
            text = df.to_string()
        elif name.endswith((".xlsx", ".xls")):
            import pandas as pd
            df = pd.read_excel(uploaded_file)
            text = df.to_string()
        elif name.endswith(".json"):
            data = json.load(uploaded_file)
            text = json.dumps(data, indent=2)
        elif name.endswith(".docx"):
            from docx import Document as DocxDoc
            doc = DocxDoc(BytesIO(uploaded_file.read()))
            text = "\n".join([p.text for p in doc.paragraphs])
    except Exception as e:
        text = f"Error: {str(e)}"
    return text


def search_context(query, raw_texts):
    query_words = query.lower().split()
    all_chunks = []
    for text in raw_texts:
        chunks = [text[i:i+800] for i in range(0, len(text), 600)]
        for chunk in chunks:
            score = sum(1 for w in query_words if w in chunk.lower())
            all_chunks.append((score, chunk))
    all_chunks.sort(reverse=True)
    return "\n\n".join([c for _, c in all_chunks[:4]])


def get_response(prompt, context, language, api_key, history=None):
    from groq import Groq
    client = Groq(api_key=api_key)
    lang_note = f"Always respond in {language}." if language != "English" else ""
    system = f"You are PapLex AI, an intelligent document assistant. Answer based on the document context provided. Be helpful and accurate. {lang_note}"
    messages = [{"role": "system", "content": system}]
    if history:
        for h in history[-4:]:
            messages.append({"role": "user", "content": h["user"]})
            messages.append({"role": "assistant", "content": h["bot"]})
    messages.append({"role": "user", "content": f"Context:\n{context[:3000]}\n\nQuestion: {prompt}"})
    resp = client.chat.completions.create(model="llama-3.3-70b-versatile", messages=messages, max_tokens=1024)
    return resp.choices[0].message.content
